def to_twos_complement(val, bits=16):
    """Converts a decimal integer to its 2's complement binary string representation."""
    if val < 0:
        val = (1 << bits) + val
    return f"{val:0{bits}b}"
def from_twos_complement(binary_str):
    """Converts a 2's complement binary string back to a signed decimal integer."""
    bits = len(binary_str)
    val = int(binary_str, 2)
    if binary_str[0] == '1':  
        val -= (1 << bits)
    return val
def execute_signed_arithmetic(op1, op2, bit_depth=16):
    """Performs 2's complement addition and displays binary and decimal steps."""
    op1_8bit = to_twos_complement(op1, 8)
    op2_8bit = to_twos_complement(op2, 8)
    op1_16bit = to_twos_complement(op1, bit_depth)
    op2_16bit = to_twos_complement(op2, bit_depth)
    mask = (1 << bit_depth) - 1
    raw_sum = int(op1_16bit, 2) + int(op2_16bit, 2)
    result_16bit = f"{(raw_sum & mask):0{bit_depth}b}"
    result_decimal = from_twos_complement(result_16bit)
    sign_op1 = int(op1_16bit[0])
    sign_op2 = int(op2_16bit[0])
    sign_res = int(result_16bit[0])
    overflow = (sign_op1 == sign_op2) and (sign_op1 != sign_res)
    print(f"Operands in 8-bit format: ")
    print(f"  Operand 1 ({op1:3d}): {op1_8bit}")
    print(f"  Operand 2 ({op2:3d}): {op2_8bit}")
    print(f"\nExecution in 16-bit 2's Complement Arithmetic:")
    print(f"    {op1_16bit}  ({op1})")
    print(f"  + {op2_16bit}  ({op2})")
    print(f"    {'-'*21}")
    print(f"    {result_16bit}  (Decimal Result: {result_decimal})")
    print(f"Arithmetic Overflow Detected: {overflow}")
    print(f"{'='*60}\n")
print("--- Verification of Arithmetic Expressions ---\n")
print("Expression a) (24 + (+17))")
execute_signed_arithmetic(24, 17)
print("Expression b) ((-25) + (-18))")
execute_signed_arithmetic(-25, -18)
